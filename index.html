function loadSite() { const input = urlInput.value.trim(); if (!input) return; let target; if (input.includes(".") && !input.includes(" ")) { target = input.startsWith("http://") || input.startsWith("https://") ? input : "https://" + input; } else { target = "https://duckduckgo.com/?q=" + encodeURIComponent(input); } window.location.assign(target); }
